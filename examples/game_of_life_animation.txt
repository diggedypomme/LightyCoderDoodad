// Conway's Game of Life for the 12x12 LightyCoder display.
// Paste this whole file into the Android app's Custom Animation box.

const reseed = () => Array.from({ length: W * H }, () => rand(1) < 0.32 ? 1 : 0);
const stepEveryFrames = 4;
const deadPauseSteps = 10;

if (!state.cells || i % 240 === 0) {
  state.cells = reseed();
  state.age = 0;
  state.deadFor = 0;
  state.stillFor = 0;
}

const cur = state.cells;
const next = new Array(W * H).fill(0);
let alive = 0;
let changes = 0;

clear(0, 0, 0);

for (let y = 0; y < H; y++) {
  for (let x = 0; x < W; x++) {
    if (cur[y * W + x]) {
      const hue = (x * 18 + y * 22 + state.age * 9) % 360;
      const [r, g, b] = hsv(hue, 0.85, 0.9);
      set(x, y, r, g, b);
      alive += 1;
    }
  }
}

if (i % stepEveryFrames !== 0) {
  return;
}

if (alive === 0) {
  state.deadFor = (state.deadFor || 0) + 1;
  if (state.deadFor >= deadPauseSteps) {
    state.cells = reseed();
    state.age = 0;
    state.deadFor = 0;
    state.stillFor = 0;
  }
  return;
}

for (let y = 0; y < H; y++) {
  for (let x = 0; x < W; x++) {
    let neighbours = 0;

    for (let dy = -1; dy <= 1; dy++) {
      for (let dx = -1; dx <= 1; dx++) {
        if (dx === 0 && dy === 0) continue;
        const xx = (x + dx + W) % W;
        const yy = (y + dy + H) % H;
        neighbours += cur[yy * W + xx];
      }
    }

    const wasAlive = cur[y * W + x] === 1;
    const isAlive = wasAlive ? neighbours === 2 || neighbours === 3 : neighbours === 3;
    next[y * W + x] = isAlive ? 1 : 0;

    if (wasAlive !== isAlive) changes += 1;
  }
}

state.age = (state.age || 0) + 1;
state.deadFor = 0;
state.stillFor = changes < 3 ? (state.stillFor || 0) + 1 : 0;

if (state.stillFor > 12 || state.age > 160) {
  state.cells = reseed();
  state.age = 0;
  state.deadFor = 0;
  state.stillFor = 0;
} else {
  state.cells = next;
}
