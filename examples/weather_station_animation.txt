// Weather station demo for the 12x12 LightyCoder display.
// Paste this whole file into the Android app's Custom Animation box.
// Scenes: simple sun, clouds, rain, lightning.

const sceneLength = 56;
const scene = Math.floor(i / sceneLength) % 4;
const tick = i % sceneLength;

function dot(x, y, r, g, b) {
  set(x, y, r, g, b);
}

function drawSimpleSun(cx, cy, pulse) {
  const bright = 0.8 + 0.2 * Math.sin(pulse / 6);
  dot(cx, cy - 2, 255 * bright, 170 * bright, 20);
  dot(cx + 2, cy, 255 * bright, 170 * bright, 20);
  dot(cx, cy + 2, 255 * bright, 170 * bright, 20);
  dot(cx - 2, cy, 255 * bright, 170 * bright, 20);
  rect(cx - 1, cy - 1, 3, 3, 255, 220 * bright, 45);
}

function drawCloud(x, y, shade) {
  const c = shade || 210;
  rect(x + 1, y + 2, 7, 2, c, c + 10, 255);
  rect(x + 2, y + 1, 2, 1, c, c + 10, 255);
  rect(x + 4, y, 3, 2, c, c + 10, 255);
  rect(x + 7, y + 1, 2, 2, c, c + 10, 255);
  dot(x, y + 3, c - 30, c - 20, 230);
  dot(x + 8, y + 3, c - 30, c - 20, 230);
}

function drawRain(frameNo) {
  for (let n = 0; n < 10; n++) {
    const x = (n * 3 + Math.floor(frameNo / 2)) % W;
    const y = (frameNo + n * 4) % H;
    dot(x, y, 40, 140, 255);
    dot(x - 1, y + 1, 20, 80, 180);
  }
}

function drawBolt(flash) {
  const color = flash ? [255, 255, 255] : [255, 230, 40];
  line(7, 4, 5, 7, color[0], color[1], color[2]);
  line(5, 7, 7, 7, color[0], color[1], color[2]);
  line(7, 7, 4, 11, color[0], color[1], color[2]);
  dot(6, 8, color[0], color[1], color[2]);
}

clear(0, 0, 0);

if (scene === 0) {
  for (let y = 0; y < H; y++) rect(0, y, W, 1, 20, 60 + y * 5, 135 + y * 5);
  drawSimpleSun(5, 4, tick);
  const level = 3 + Math.floor((Math.sin(tick / 8) + 1) * 2);
  rect(9, 3, 1, 7, 80, 80, 95);
  rect(8, 9, 3, 2, 80, 80, 95);
  for (let n = 0; n < level; n++) rect(9, 8 - n, 1, 1, 255, 90 + n * 25, 30);
  rect(8, 10, 3, 1, 255, 80, 20);
}

if (scene === 1) {
  for (let y = 0; y < H; y++) rect(0, y, W, 1, 15, 45 + y * 4, 95 + y * 6);
  const x1 = -7 + (tick % 22);
  const x2 = 12 - (tick % 20);
  drawCloud(x1, 2, 210);
  drawCloud(x2, 6, 170);
  dot(1, 1, 255, 255, 190);
  dot(10, 2, 180, 220, 255);
}

if (scene === 2) {
  for (let y = 0; y < H; y++) rect(0, y, W, 1, 5, 18 + y * 2, 45 + y * 5);
  drawCloud(1, 1, 145);
  drawRain(tick);
  if (tick % 10 < 4) rect(1, 10, 10, 1, 30, 80, 150);
}

if (scene === 3) {
  const flash = tick % 18 < 3 || tick % 31 === 0;
  rect(0, 0, W, H, flash ? 130 : 8, flash ? 130 : 10, flash ? 150 : 28);
  drawCloud(1, 1, flash ? 230 : 110);
  drawRain(tick + 8);
  drawBolt(flash);
}
